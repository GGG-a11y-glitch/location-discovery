'night' : 'day'; document.body.className = `theme-${state.currentTheme}`; document.getElementById('time-badge').innerText = isNight ? "🌙 Notte" : "☀️ Giorno"; document.getElementById('hero-title').innerText = isNight ? "Night Discovery" : "City Explorer"; }; // --- 3. CALCOLO DISTANZA --- const getDistance = (lat1, lon1, lat2, lon2) => { const R = 6371; const dLat = (lat2 - lat1) * Math.PI / 180; const dLon = (lon2 - lon1) * Math.PI / 180; const a = Math.sin(dLat/2) * Math.sin(dLat/2) + Math.cos(lat1 * Math.PI / 180) * Math.cos(lat2 * Math.PI / 180) * Math.sin(dLon/2) * Math.sin(dLon/2); return R * (2 * Math.atan2(Math.sqrt(a), Math.sqrt(1-a))); }; // --- 5. RENDERING --- const renderLocations = () => { const grid = document.getElementById('location-grid'); const skeleton = document.getElementById('skeleton-grid'); if (!grid) return; // Filtra per tema const nearby = POI_DATABASE.filter(poi => state.currentTheme === 'night' ? true : poi.type === 'outdoor' ); skeleton.classList.add('hidden'); grid.classList.remove('hidden'); grid.innerHTML = ''; nearby.forEach(poi => { let distText = "--"; let isNear = false; if (state.userCoords) { const dist = getDistance(state.userCoords.lat, state.userCoords.lon, poi.lat, poi.lon); distText = `${dist.toFixed(2)} km`; isNear = dist < 0.5; if (isNear && state.isRadarActive && !state.vibratedPOIs.has(poi.id)) { if (navigator.vibrate) navigator.vibrate([100, 50, 100]); state.vibratedPOIs.add(poi.id); } } const card = document.createElement('div'); card.className = `card ${isNear && state.isRadarActive ? 'near' : ''}`; card.innerHTML = `

${poi.name}

${poi.desc}

${distText}
`; grid.appendChild(card); }); if (state.userCoords) { document.getElementById('hero-subtitle').innerText = `${nearby.length} luoghi trovati vicino a te`; } }; const toggleRadar = () => { state.isRadarActive = !state.isRadarActive; const btn = document.getElementById('radar-toggle'); btn.classList.toggle('active'); btn.querySelector('.status-label').innerText = state.isRadarActive ? 'ON' : 'OFF'; state.vibratedPOIs.clear(); renderLocations(); }; const sharePOI = (name) => { if (navigator.share) { navigator.share({ title: 'Discovery', text: `Andiamo a ${name}?`, url: window.location.href }); } }; const initParallax = () => { window.addEventListener('scroll', () => { const bg = document.getElementById('hero-bg'); if(bg) bg.style.transform = `translateY(${window.pageYOffset * 0.4}px)`; }); }; // Avvio manuale init();
